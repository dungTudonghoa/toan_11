\begin{theorybox}
\textbf{Kiến thức trọng tâm:} Phương trình $\sin x=a$, $\cos x=a$,
$\tan x=a$, $\cot x=a$; vận dụng công thức nghiệm và đếm nghiệm trên một khoảng.
Trong tài liệu, $k,l\in\mathbb Z$; những góc không ghi $^\circ$ được hiểu theo radian.
\end{theorybox}

\sectitle{1}{Phương trình $\sin x=a$}
Nếu $|a|>1$, phương trình vô nghiệm; nếu $-1\le a\le1$, phương trình có nghiệm.
\begin{minipage}[c]{.72\linewidth}
\begin{itemize}
 \item $\sin x=1\iff x=\dfrac{\pi}{2}+2k\pi$.
 \item $\sin x=-1\iff x=-\dfrac{\pi}{2}+2k\pi$.
 \item $\sin x=0\iff x=k\pi$.
 \item $\sin x=\sin\alpha\iff\begin{cases}x=\alpha+2k\pi,\\x=\pi-\alpha+2k\pi.\end{cases}$
 \item $\sin x=\sin\beta^\circ\iff\begin{cases}x=\beta^\circ+k360^\circ,\\x=(180^\circ-\beta^\circ)+k360^\circ.\end{cases}$
\end{itemize}
\end{minipage}\hfill
\begin{minipage}[c]{.26\linewidth}\centering\unitcircle\end{minipage}
\begin{theorybox}
Với $\alpha=\arcsin a\in[-\pi/2,\pi/2]$ ($-1\le a\le1$),
\[
 \boxed{\sin x=a\iff\begin{cases}
 x=\arcsin a+2k\pi,\\x=\pi-\arcsin a+2k\pi.
 \end{cases}}
\]
$\sin f(x)=\sin g(x)\iff
 \begin{cases}f(x)=g(x)+2k\pi,\\f(x)=\pi-g(x)+2k\pi.\end{cases}$

Ví dụ: $\arcsin\dfrac12=\dfrac\pi6$,\quad
$\arcsin\left(-\dfrac{\sqrt3}{2}\right)=-\dfrac\pi3$.
\end{theorybox}
\exampletitle{01 [TDM31]}{Giải các phương trình lượng giác sau:}
\begingroup\compactproblems
\begin{multicols}{2}
\begin{enumerate}
 \item $\sin x=\sin\dfrac{\pi}{4}$.
 \item $\sin x=-\dfrac{\sqrt3}{2}$.
 \item $\sin x=2$.
 \item $\sin x=\dfrac25$.
 \item $\sin(x+60^\circ)=\dfrac12$.
 \item $\sin(x+2)=\sin(2x-1)$.
 \item $\sin x=\cos x$.
 \item $\sin x=-\cos x$.
 \item $\sin(2x+3)=-\sin x$.
\end{enumerate}
\end{multicols}
\endgroup

\sectitle{2}{Phương trình $\cos x=a$}
Nếu $|a|>1$, phương trình vô nghiệm; nếu $-1\le a\le1$, phương trình có nghiệm.
\begin{minipage}[c]{.72\linewidth}
\begin{itemize}
 \item $\cos x=1\iff x=2k\pi$.
 \item $\cos x=-1\iff x=\pi+2k\pi$.
 \item $\cos x=0\iff x=\dfrac{\pi}{2}+k\pi$.
 \item $\cos x=\cos\alpha\iff x=\pm\alpha+2k\pi$.
 \item $\cos x=\cos\beta^\circ\iff x=\pm\beta^\circ+k360^\circ$.
\end{itemize}
\end{minipage}\hfill
\begin{minipage}[c]{.26\linewidth}\centering\unitcircle\end{minipage}
\begin{theorybox}
Với $\alpha=\arccos a\in[0,\pi]$ ($-1\le a\le1$),
\[
 \boxed{\cos x=a\iff x=\pm\arccos a+2k\pi.}
\]
$\cos f(x)=\cos g(x)\iff f(x)=\pm g(x)+2k\pi$.

Ví dụ: $\arccos 0=\dfrac\pi2$,\quad $\arccos\dfrac12=\dfrac\pi3$.
\end{theorybox}
\exampletitle{01 [TDM31]}{Giải các phương trình lượng giác sau:}
\begingroup\compactproblems
\begin{multicols}{2}
\begin{enumerate}
 \item $\cos x=\dfrac12$.
 \item $\cos x=\cos\dfrac{5\pi}{7}$.
 \item $\cos x=-3$.
 \item $\cos(x+30^\circ)=-\dfrac12$.
 \item $\cos x=\dfrac14$.
 \item $\cos(2x-3)=\cos x$.
 \item $\cos x=-\cos(x+1)$.
 \item $\cos 2x=\sin x$.
 \item $\cos 3x=-\sin x$.
\end{enumerate}
\end{multicols}
\endgroup

\sectitle{3}{Phương trình $\tan x=a$}
Với mọi $a\in\mathbb R$, phương trình $\tan x=a$ luôn có nghiệm.
\begin{minipage}[c]{.64\linewidth}
\begin{itemize}
 \item $\tan x=\tan\alpha\iff x=\alpha+k\pi$.
 \item $\tan x=\tan\beta^\circ\iff x=\beta^\circ+k180^\circ$.
 \item $\tan f(x)=\tan g(x)\iff
 \begin{cases}f(x)\ne\dfrac\pi2+l\pi,\\f(x)=g(x)+k\pi.\end{cases}$
 \item $\alpha=\arctan a\in(-\pi/2,\pi/2)$.
\end{itemize}
\end{minipage}\hfill
\begin{minipage}[c]{.35\linewidth}\centering\tanpicture\end{minipage}
\begin{theorybox}
\[
 \boxed{\tan x=a\iff x=\arctan a+k\pi.}
\]
Ví dụ: $\arctan 0=0$,\quad $\arctan(-\sqrt3)=-\dfrac\pi3$.
\end{theorybox}
\exampletitle{01 [TDM31]}{Giải các phương trình lượng giác sau:}
\begingroup\compactproblems
\begin{multicols}{2}
\begin{enumerate}
 \item $\tan x=\tan\dfrac\pi5$.
 \item $\tan x=-\sqrt3$.
 \item $\tan x=4$.
 \item $\tan(x+60^\circ)=1$.
 \item $\tan(2x+3)=\tan x$.
 \item $\tan 2x\cdot\tan x=-1$.
\end{enumerate}
\end{multicols}
\endgroup

\sectitle{4}{Phương trình $\cot x=a$}
Với mọi $a\in\mathbb R$, phương trình $\cot x=a$ luôn có nghiệm.
\begin{minipage}[c]{.64\linewidth}
\begin{itemize}
 \item $\cot x=\cot\alpha\iff x=\alpha+k\pi$.
 \item $\cot x=\cot\beta^\circ\iff x=\beta^\circ+k180^\circ$.
 \item $\cot f(x)=\cot g(x)\iff
 \begin{cases}f(x)\ne l\pi,\\f(x)=g(x)+k\pi.\end{cases}$
 \item $\alpha=\operatorname{arccot}a\in(0,\pi)$.
\end{itemize}
\end{minipage}\hfill
\begin{minipage}[c]{.35\linewidth}\centering\cotpicture\end{minipage}
\begin{theorybox}
\[
 \boxed{\cot x=a\iff x=\operatorname{arccot}a+k\pi.}
\]
Thông thường:
\[
 \cot x=0\iff x=\frac\pi2+k\pi;\qquad
 \cot x=a\ne0\iff x=\arctan\frac1a+k\pi.
\]
Ví dụ: $\operatorname{arccot}0=\dfrac\pi2$,\quad
$\operatorname{arccot}(-\sqrt3)=\dfrac{5\pi}{6}$.
\end{theorybox}
\exampletitle{01 [TDM31]}{Giải các phương trình lượng giác sau:}
\begingroup\compactproblems
\begin{multicols}{2}
\begin{enumerate}
 \item $\cot x=\sqrt3$.
 \item $\cot(x-30^\circ)=0$.
 \item $\cot x=5$.
 \item $\cot(2x-1)=\cot x$.
\end{enumerate}
\end{multicols}
\endgroup

\sectitle{5}{Kĩ năng đếm nghiệm của một phương trình lượng giác}
\exampletitle{01 [TDM41]}{Đếm số nghiệm thực $x$ của phương trình:}
\begingroup\compactproblems
\begin{enumerate}
 \item $\cos x=\dfrac12$ trên nửa khoảng $\left(0;\dfrac{7\pi}{3}\right]$.
 \item $\cos x=-0{,}8$ trên nửa khoảng $[-2025;2029)$.
\end{enumerate}
\endgroup
\exampletitle{02 [TDM42]}{Tìm điều kiện của tham số $m$ để phương trình:}
\begingroup\compactproblems
\begin{enumerate}
 \item $\sin x=m$ có đúng $3$ nghiệm trên đoạn $\left[0;\dfrac{17\pi}{6}\right]$.
 \item $\cos 2x=m$ có đúng $67$ nghiệm trên nửa khoảng $\left[0;\dfrac{100\pi}{3}\right)$.
\end{enumerate}
\endgroup
